 202: def qimage_to_numpy(qimg: QImage) -> np.ndarray:
 203:     """Chuyển QImage → numpy array (BGR)."""
 204:     qimg = qimg.convertToFormat(QImage.Format_RGB32)
 205:     width = qimg.width()
 206:     height = qimg.height()
 207: 
 208:     ptr = qimg.bits()
 209:     ptr.setsize(height * width * 4)
 210:     arr = np.frombuffer(ptr, dtype=np.uint8).reshape((height, width, 4))
 211: 
 212:     # BGRA → BGR
 213:     bgr = cv2.cvtColor(arr, cv2.COLOR_BGRA2BGR)
 214:     return bgr

 217: def numpy_to_qpixmap(image: np.ndarray) -> QPixmap:
 218:     """Chuyển numpy array (BGR) → QPixmap."""
 219:     if len(image.shape) == 2:
 220:         h, w = image.shape
 221:         qimg = QImage(image.data, w, h, w, QImage.Format_Grayscale8)
 222:     else:
 223:         rgb = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
 224:         h, w, ch = rgb.shape
 225:         bytes_per_line = ch * w
 226:         qimg = QImage(rgb.data, w, h, bytes_per_line, QImage.Format_RGB888)
 227:     return QPixmap.fromImage(qimg.copy())
